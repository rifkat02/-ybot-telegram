Нажатие на кнопку "Купить сразу" function buyNow() { if (!isAuctionActive) return; tg.showConfirm(`Вы уверены, что хотите купить этот лот за ${buyNowPrice.toLocaleString('ru-RU')} ₽?`, (confirmed) => { if (confirmed) { showPaymentModal(); } }); } // 2. Показ всплывающего окна с выбором способа оплаты function showPaymentModal() { const modalHtml = `

Сумма к оплате: ${buyNowPrice.toLocaleString('ru-RU')} ₽

Номер: +7 (900) 000-00-00

Получатель: Ваше Имя М.

Переведите точную сумму и нажмите кнопку ниже:

Безопасная оплата картой любой страны или через Tinkoff Pay / СберПэй.

`; document.body.insertAdjacentHTML('beforeend', modalHtml); } // 3. Переключение между табами оплаты function switchPayTab(type) { const sbpBlock = document.getElementById('paySbpBlock'); const onlineBlock = document.getElementById('payOnlineBlock'); const sbpBtn = document.getElementById('tabSbpBtn'); const onlineBtn = document.getElementById('tabOnlineBtn'); if (type === 'sbp') { sbpBlock.style.display = 'block'; onlineBlock.style.display = 'none'; sbpBtn.classList.add('active'); onlineBtn.classList.remove('active'); } else { sbpBlock.style.display = 'none'; onlineBlock.style.display = 'block'; sbpBtn.classList.remove('active'); onlineBtn.classList.add('active'); } } // 4. Подтверждение оплаты по СБП function confirmSbpPayment() { closeModal(); endAuction("🎉 Лот выкуплен! Ожидается подтверждение перевода."); tg.HapticFeedback.notificationOccurred('success'); // Передаем данные боту const orderData = { action: 'buy_now', pay_method: 'СБП / Перевод', price: buyNowPrice, title: document.getElementById('lotTitle').innerText, user: tg.initDataUnsafe?.user?.username ? `@${tg.initDataUnsafe.user.username}` : (tg.initDataUnsafe?.user?.first_name || 'Покупатель') }; tg.sendData(JSON.stringify(orderData)); } // 5. Переход на внешнюю ссылку эквайринга function payViaGateway() { // Укажите вашу ссылку на форму оплаты (например, ЮKassa или CloudPayments) const paymentUrl = `https://yoomoney.ru/pay?amount=${buyNowPrice}`; tg.openLink(paymentUrl); // Открывает платежную страницу } // Закрытие модального окна function closeModal() { const modal = document.getElementById('payModal'); if (modal) modal.remove(); }
